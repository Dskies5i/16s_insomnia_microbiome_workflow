# ============================================================
# NOMBRE: 01_build_phyloseq.R
# DESCRIPCIÓN: Construcción, filtrado y normalización del objeto 
#              phyloseq global a partir de las matrices unificadas 
#              de QIIME 2 (VSEARCH 99%) y metadatos maestros.
#
# ¿QUÉ HACE ESTE SCRIPT?
#    1. Lee la tabla OTU/ASV unificada, la taxonomía SILVA y la 
#       tabla maestra de metadatos desde data/pre_processed/unified/.
#    2. Construye el objeto phyloseq integrando conteos, taxonomía y muestras.
#    3. Aplica filtros de calidad (elimina taxones no deseados como 'g__other' 
#       o 'Unassigned' y remueve sumas en cero).
#    4. Fija y estandariza los niveles de los grupos de estudio (Control vs. Insomnia).
#    5. Guarda el objeto phyloseq resultante y exporta las tablas filtradas 
#       de soporte a outputs/models/ y outputs/tables/.
#
# ARCHIVOS DE SALIDA (OUTPUTS):
#    - outputs/models/physeq_filtered.rds       (Objeto phyloseq completo y filtrado en R)
#    - outputs/tables/tax_filtered.csv          (Tabla taxonómica limpia y filtrada)
#    - outputs/tables/otu_filtered.csv          (Matriz de conteos filtrada)
#    - outputs/tables/metadata_filtered.csv     (Tabla maestra de metadatos filtrada)
#
# USO: Rscript 1_meta_analysis/01_build_phyloseq.R
#
# ENVIRONMENT: conda activate qiime2-amplicon-2024.5
# phyloseq is instaleld inside "qiime2-amplicon-2024.5" environment
#
# WORKS? -- YES
# ============================================================
source("1_meta_analysis/00_setup.R")
library(phyloseq)

ps <- build_phyloseq()

if ("Genus" %in% colnames(tax_table(ps))) {
  ps <- subset_taxa(ps, !grepl("g__other|Unassigned", tax_table(ps)[, "Genus"]))
}
ps <- prune_taxa(taxa_sums(ps) > 0, ps)
ps <- prune_samples(sample_sums(ps) > 0, ps)
ps <- enforce_group_levels(ps, cfg$columns$group, cfg$groups$levels)

save_rds_safe(ps, file.path(OUT_DIR, "models/physeq_filtered.rds"))

tax  <- as.data.frame(tax_table(ps))    %>% tibble::rownames_to_column("OTU")
otu  <- as.data.frame(otu_table(ps))    %>% tibble::rownames_to_column("OTU")
meta <- as.data.frame(sample_data(ps))  %>% tibble::rownames_to_column("Sample")

write_csv_safe(tax,  file.path(OUT_DIR, "tables/tax_filtered.csv"))
write_csv_safe(otu,  file.path(OUT_DIR, "tables/otu_filtered.csv"))
write_csv_safe(meta, file.path(OUT_DIR, "tables/metadata_filtered.csv"))

cat("✅ Objeto phyloseq construido y guardado en outputs/models/\n")
cat("   Taxa:", ntaxa(ps), "| Samples:", nsamples(ps), "\n")

# ============================================================
# NOMBRE: 02_alpha_diversity.R
# DESCRIPCIÓN: Calcula la diversidad alfa (Shannon, Simpson, etc.) 
#              a partir del objeto phyloseq filtrado, genera gráficos 
#              de violín/caja comparativos entre grupos y ejecuta pruebas 
#              estadísticas (t-test).
#
# ¿QUÉ HACE ESTE SCRIPT?
#    1. Carga el objeto phyloseq filtrado desde outputs/models/.
#    2. Estima los índices de diversidad alfa configurados en config.yaml.
#    3. Combina los valores de diversidad con los metadatos clínicos.
#    4. Genera y guarda gráficos combinados (Shannon y Simpson) en outputs/figures/.
#    5. Realiza pruebas estadísticas t-test y exporta los logs a outputs/logs/.
#
# ARCHIVOS DE SALIDA (OUTPUTS):
#    - outputs/tables/alpha_diversity.csv        (Tabla completa con métricas de diversidad por muestra)
#    - outputs/figures/alpha_shannon_simpson.png (Gráfico combinado en formato violín/boxplots)
#    - outputs/logs/alpha_stats.txt              (Resultados detallados de las pruebas estadísticas t-test)
#
# USO: Rscript 1_meta_analysis/02_alpha_diversity.R
#
# PARA ESTE SCRIPT EN TERMINAL: Rscript -e "install.packages(c('patchwork', 'ggpubr', 'Boruta', 'pheatmap', 'RColorBrewer'), repos='https://cloud.r-project.org')"
# 
# ENVIRONMENT: conda activate qiime2-amplicon-2024.5
#
# WORKS? -- YES 
#============================================================
source("1_meta_analysis/00_setup.R")
library(phyloseq)
library(patchwork)

# 1. Cargar el objeto phyloseq filtrado
ps_path <- file.path(OUT_DIR, "models/physeq_filtered.rds")
if (!file.exists(ps_path)) {
  stop("❌ ERROR: No se encontró el archivo physeq_filtered.rds. Ejecuta primero 01_build_phyloseq.R")
}
ps <- readRDS(ps_path)
ps <- prune_taxa(taxa_sums(ps) > 0, ps)

# 2. Estimar diversidad alfa con base en las medidas del archivo de configuración
alpha_measures <- cfg$analysis$alpha$measures
alpha_tab <- estimate_richness(ps, measures = alpha_measures)
alpha_tab$Sample <- rownames(alpha_tab)

# 3. Incorporar metadatos (Grupo de estudio)
meta <- data.frame(sample_data(ps)) %>% rownames_to_column("Sample")
group_col <- cfg$columns$group

alpha_tab <- alpha_tab %>%  
  left_join(meta %>% select(Sample, Group = all_of(group_col)), by = "Sample")

# Forzar el orden de los niveles de los grupos según la configuración
groups_levels <- cfg$groups$levels
alpha_tab$Group <- factor(alpha_tab$Group, levels = groups_levels)

# Guardar tabla completa de diversidad alfa
write_csv_safe(alpha_tab, file.path(OUT_DIR, "tables/alpha_diversity.csv"))

# 4. Generación de gráficos de violín y cajas
cols <- unlist(cfg$groups$colors)

p_shannon <- ggplot(alpha_tab, aes(x = Group, y = Shannon, fill = Group)) +
  geom_violin(trim = FALSE, alpha = 0.7) +
  geom_boxplot(width = 0.1, outlier.shape = NA, color = "black") +
  stat_summary(fun = mean, geom = "point", shape = 21, size = 2.5, fill = "white") +
  scale_fill_manual(values = cols) +
  theme_minimal() + 
  theme(legend.position = "none", plot.title = element_text(size = 11, face = "bold")) +
  labs(title = "Shannon Diversity", x = "", y = "Shannon Index")

p_simpson <- ggplot(alpha_tab, aes(x = Group, y = Simpson, fill = Group)) +
  geom_violin(trim = FALSE, alpha = 0.7) +
  geom_boxplot(width = 0.1, outlier.shape = NA, color = "black") +
  stat_summary(fun = mean, geom = "point", shape = 21, size = 2.5, fill = "white") +
  scale_fill_manual(values = cols) +
  theme_minimal() + 
  theme(legend.position = "none", plot.title = element_text(size = 11, face = "bold")) +
  labs(title = "Simpson Diversity", x = "", y = "Simpson Index")

# Guardar gráfico combinado mediante patchwork
save_plot(p_shannon + p_simpson, file.path(OUT_DIR, "figures/alpha_shannon_simpson.png"), w = 9, h = 4)

# 5. Ejecución de pruebas estadísticas (t-test) y guardado de logs
sh_t <- t.test(Shannon ~ Group, data = alpha_tab)
si_t <- t.test(Simpson ~ Group, data = alpha_tab)

stats_output <- c(
  "==========================================",
  " RESUMEN DE PRUEBAS ESTADÍSTICAS - DIVERSIDAD ALFA",
  "==========================================",
  "",
  "--- Shannon Index (t-test) ---",
  capture.output(print(sh_t)),
  "",
  "--- Simpson Index (t-test) ---",
  capture.output(print(si_t)),
  "=========================================="
)

write_text_safe(stats_output, file.path(OUT_DIR, "logs/alpha_diversity_stats.txt"))

cat("✅ Análisis de diversidad alfa completado con éxito. Resultados guardados en outputs/\n")

# 6. Previsualización rápida en consola
cat("\n==========================================")
cat("\n RESUMEN DE DIVERSIDAD ALFA (Primeras filas)")
cat("\n==========================================\n")
print(head(alpha_tab, 5))
cat("---------------------\nResultados rápidos de las pruebas t-test:\n")
print(sh_t)
print(si_t)
cat("==========================================\n\n")

  alpha <- readr::read_csv(file.path(out_dir, "tables/alpha_diversity.csv"),
                           show_col_types = FALSE)
  required_cols <- c("Sample", "Shannon", "Simpson", "Group")
  missing <- setdiff(required_cols, colnames(alpha))
  if (length(missing) > 0) {
    cat("❌ Columnas faltantes:", paste(missing, collapse = ", "), "\n")
    all_ok <- FALSE
  } else {
    cat(sprintf("✅ alpha_diversity.csv: %d muestras, %d columnas\n",

    na_sh  <- sum(is.na(alpha$Shannon))
    na_si  <- sum(is.na(alpha$Simpson))
    if (na_sh > 0 || na_si > 0) {
      cat(sprintf("⚠️ NA detectados — Shannon: %d, Simpson: %d\n", na_sh, na_si))
      all_ok <- FALSE
    } else {
      cat("   ✅ Sin NAs en Shannon / Simpson\n")
    }
    
    # Group counts
    tab <- table(alpha$Group)
    cat("   Group counts: ",
        paste(names(tab), "=", as.integer(tab), collapse = ", "), "\n")
  }
}, error = function(e) {
  cat("❌ Error leyendo alpha_diversity.csv:", e$message, "\n")
  all_ok <<- FALSE
})

# NOMBRE: 04_fb_ratio.R
# DESCRIPCIÓN: Cálculo de la razón Firmicutes (Bacillota) a 
#              Bacteroidota por muestra, pruebas estadísticas y 
#              generación de boxplots comparativos.
#
# ¿QUÉ HACE ESTE SCRIPT?
#    1. Carga el phyloseq y extrae perfiles taxonómicos a nivel Phylum.
#    2. Calcula la razón F:B para cada muestra de manera robusta.
#    3. Genera un gráfico de cajas con valores de dispersión (jitter).
#    4. Realiza t-test y exporta tanto la tabla de valores como los logs.
#
# ARCHIVOS DE SALIDA (OUTPUTS):
#    - outputs/tables/fb_ratio.csv          (Valores de la razón F:B por muestra)
#    - outputs/figures/fb_ratio.png         (Gráfico de cajas comparativo)
#    - outputs/logs/fb_ratio_stats.txt      (Resultados estadísticos de la prueba t)
#
# USO: Rscript 1_meta_analysis/04_fb_ratio.R
#
# ENVIRONMENT: conda activate qiime2-amplicon-2024.5
#
# WORKS? -- ???
# ============================================================
# EN TERMINAL: Rscript -e "install.packages('ggpubr', repos='https://cloud.r-project.org')"
# OR: conda install -c conda-forge r-ggpubr -y


source("1_meta_analysis/00_setup.R")
library(phyloseq)
library(ggpubr)

ps <- readRDS(file.path(OUT_DIR, "models/physeq_filtered.rds"))

otu <- as.data.frame(otu_table(ps), check.names = FALSE)
if (!taxa_are_rows(ps)) otu <- t(otu)
tax <- as.data.frame(tax_table(ps), check.names = FALSE)
meta <- as.data.frame(sample_data(ps))

ph_a <- cfg$analysis$fb_ratio$phylum_a  # "Firmicutes"
ph_b <- cfg$analysis$fb_ratio$phylum_b  # "Bacteroidota"

# Cálculo de la razón F:B por muestra
bf_ratio <- apply(otu, 2, function(x) {
  b <- sum(x[which(tax[, "Phylum"] == ph_b)], na.rm = TRUE)
  f <- sum(x[which(tax[, "Phylum"] == ph_a)], na.rm = TRUE)
  ifelse(b == 0, NA, f / b)
})

# Alineación limpia de muestras entre OTU table y metadatos
common_samples <- intersect(names(bf_ratio), rownames(meta))
group_col <- cfg$columns$group

# Extracción robusta garantizando vector plano de caracteres
group_vals <- meta[common_samples, group_col]
if (is.data.frame(group_vals)) group_vals <- group_vals[[1]]
group_vals <- trimws(as.character(group_vals))

df <- data.frame(
  Sample = common_samples,
  FB_ratio = bf_ratio[common_samples],
  Group_raw = group_vals,
  stringsAsFactors = FALSE
)

# Mapeo explícito tanto para códigos numéricos ("1"/"2") como texto ("Control"/"Insomnia")
df$Group <- ifelse(df$Group_raw == "1" | tolower(df$Group_raw) == "control", "Control",
                   ifelse(df$Group_raw == "2" | tolower(df$Group_raw) == "insomnia", "Insomnia", NA))

# Filtrado de NAs y asignación de niveles de factor
df <- df[!is.na(df$FB_ratio) & !is.na(df$Group), ]
df$Group <- factor(df$Group, levels = cfg$groups$levels)

cat("\nConteo de muestras por grupo (mapeo correcto):\n")
print(table(df$Group))

write_csv_safe(df, file.path(OUT_DIR, "tables/fb_ratio.csv"))

# Gráfico
cols <- unlist(cfg$groups$colors)

p <- ggplot(df, aes(x = Group, y = FB_ratio, fill = Group)) +
  geom_boxplot(outlier.shape = NA, alpha = 0.6) +
  geom_jitter(width = 0.2, size = 1.2, alpha = 0.5) +
  stat_compare_means(method = "t.test", label = "p.format",
                     #label.y = max(df$FB_ratio, na.rm = TRUE) * 1.05) +
                     label.y = max(df$FB_ratio, na.rm = TRUE) * 1.10) +
  scale_fill_manual(values = cols) +
  theme_minimal() +
  labs(title = "Firmicutes:Bacteroidota ratio", y = "F:B ratio")

save_plot(p, file.path(OUT_DIR, "figures/fb_ratio.png"), w = 6, h = 4)

# Estadísticas
st <- t.test(FB_ratio ~ Group, data = df)
write_text_safe(capture.output(st), file.path(OUT_DIR, "logs/fb_ratio_stats.txt"))

cat("\n✅ Análisis de ratio Firmicutes:Bacteroidota completado con éxito.\n")

cat("\n==========================================")
cat("\n RESUMEN DE LA RAZÓN FIRMICUTES:BACTEROIDOTA")
cat("\n==========================================\n")
print(summary(df$FB_ratio))
cat("\nResultados de la prueba t:\n")
print(st)
cat("==========================================\n\n")

# NOMBRE: 05_boruta.R
# DESCRIPCIÓN: Selección de características (features) importantes 
#              mediante el algoritmo de Machine Learning Boruta 
#              para discriminar entre grupos de estudio.
#
# ¿QUÉ HACE ESTE SCRIPT?
#    1. Prepara las matrices de metadatos y abundancia de ASVs.
#    2. Ejecuta el algoritmo Boruta incorporando covariables.
#    3. Extrae los ASVs relevantes identificados como importantes.
#    4. Anota taxonómicamente los ASVs seleccionados y genera gráfico.
#
# ARCHIVOS DE SALIDA (OUTPUTS):
#    - outputs/boruta/boruta_model.rds                  (Modelo Boruta guardado)
#    - outputs/boruta/boruta_selected_otu.csv           (Matriz de ASVs seleccionados)
#    - outputs/boruta/boruta_selected_otus_with_taxonomy.csv (Taxonomía de ASVs clave)
#    - outputs/boruta/boruta_importance.png             (Gráfico de importancia de variables)
#
# USO: Rscript 1_meta_analysis/05_boruta.R
#
# ENVIRONMENT: conda activate qiime2-amplicon-2024.5
#
# WORKS? -- YES !!!! 
#
# REQUIRED INSTALL: conda install -c conda-forge r-boruta -y 
#  (IN TERMINAL - INSIDE ENVIRONMENT)
# ============================================================

source("1_meta_analysis/00_setup.R")
library(Boruta)
library(tibble)

out_dir <- file.path(OUT_DIR, "boruta")
dir_create(out_dir)

# Cargar phyloseq filtrado
ps <- readRDS(file.path(OUT_DIR, "models/physeq_filtered.rds"))

# Extraer OTUs y metadatos limpios
otu_mat <- as.data.frame(otu_table(ps), check.names = FALSE)
if (!taxa_are_rows(ps)) otu_mat <- t(otu_mat)
X <- as.data.frame(t(otu_mat))

# Aplicar prefijo seguro para evitar errores en fórmulas de R si los IDs son numéricos
otu_prefix <- cfg$differential$boruta$otu_prefix
colnames(X) <- paste0(otu_prefix, colnames(X))

meta <- as.data.frame(sample_data(ps))

target <- cfg$columns$group
covs <- cfg$differential$boruta$covariates

# Ajustar tipos de datos
meta[[target]] <- as.factor(meta[[target]])
valid_covs <- intersect(covs, colnames(meta))
for (cc in valid_covs) {
  meta[[cc]] <- as.factor(meta[[cc]])
}

Y <- meta[[target]]
X_full <- cbind(meta[, valid_covs, drop = FALSE], X)

df_boruta <- data.frame(Group = Y, X_full)
colnames(df_boruta)[1] <- "Group"

# Ejecutar Boruta
boruta_res <- Boruta(
  Group ~ .,
  data = df_boruta,
  doTrace = cfg$differential$boruta$doTrace,
  maxRuns = cfg$differential$boruta$maxRuns
)

saveRDS(boruta_res, file.path(out_dir, "boruta_model.rds"))

# Extracción de variables seleccionadas
final_vars <- getSelectedAttributes(boruta_res, withTentative = FALSE)
final_otus <- final_vars[grepl(paste0("^", otu_prefix), final_vars)]

if(length(final_otus) > 0) {
  selected_otu_table <- X[, final_otus, drop = FALSE]
  write_csv_safe(
    selected_otu_table %>% tibble::rownames_to_column("Sample"),
    file.path(out_dir, "boruta_selected_otu.csv")
  )
  
  # Anotación taxonómica (quitando el prefijo temporal para hacer el join)
  taxonomy <- as.data.frame(tax_table(ps)) %>% tibble::rownames_to_column("OTU_ID")
  #boruta_otus_df <- data.frame(OTU_ID = gsub(otu_prefix, "", final_otus))
  #AFTER
  boruta_otus_df <- data.frame(OTU_ID = sub(paste0("^", otu_prefix), "", final_otus))
  boruta_annot <- dplyr::left_join(boruta_otus_df, taxonomy, by = "OTU_ID") %>%
    dplyr::select(OTU_ID, Kingdom, Phylum, Class, Order, Family, Genus, Species)
  
  write_csv_safe(boruta_annot, file.path(out_dir, "boruta_selected_otus_with_taxonomy.csv"))
} else {
  warning("⚠️ Boruta no seleccionó ningún ASV con los parámetros actuales.")
}

cat("✅ Selección de características con Boruta finalizada con éxito.\n")
cat(sprintf("\nASVs seleccionados como importantes: %d\n\n", length(final_otus)))




# ============================================================
# VISUALIZACIÓN MEJORADA DE BORUTA (ggplot2 horizontal)
# ============================================================

library(ggplot2)

# 1. Extraer importancia de todas las decisiones
imp_mat <- attStats(boruta_res)
imp_mat$Feature <- rownames(imp_mat)

# 2. Filtrar solo confirmados, tentativas y sombras de referencia
imp_filtered <- imp_mat[imp_mat$decision %in% c("Confirmed", "Tentative") | 
                          grepl("^shadow", imp_mat$Feature), ]

# 3. Mapear con la taxonomía para nombres legibles
taxonomy <- as.data.frame(tax_table(ps)) %>% tibble::rownames_to_column("OTU_ID")
imp_filtered$OTU_ID <- gsub(otu_prefix, "", imp_filtered$Feature)

imp_filtered <- dplyr::left_join(imp_filtered, taxonomy, by = "OTU_ID")

# Crear etiqueta legible (Género/Especie o la sombra)
imp_filtered$Label <- ifelse(grepl("^shadow", imp_filtered$Feature), 
                             imp_filtered$Feature,
                             ifelse(!is.na(imp_filtered$Genus) & imp_filtered$Genus != "g__other",
                                    paste0(imp_filtered$Genus, " (", substr(imp_filtered$OTU_ID, 1, 6), ")"),
                                    paste0("ASV_", substr(imp_filtered$OTU_ID, 1, 8))))

# Ordenar por importancia media
imp_filtered$Label <- factor(imp_filtered$Label, levels = imp_filtered$Label[order(imp_filtered$meanImp)])

# 4. Graficar con ggplot2
p_boruta <- ggplot(imp_filtered, aes(x = meanImp, y = Label, fill = decision)) +
  geom_col(alpha = 0.85, color = "black", linewidth = 0.3) +
  scale_fill_manual(values = c("Confirmed" = "#2b5c8f", 
                               "Tentative" = "#e6ab02", 
                               "Shadow"    = "#7570b3",
                               "Rejected"  = "#d95f02")) +
  theme_minimal(base_size = 12) +
  labs(
    title = "Boruta Feature Importance (Confirmed ASVs)",
    x = "Mean Importance Score",
    y = "Taxon / Feature",
    fill = "Decision"
  ) +
  theme(
    axis.text.y = element_text(size = 10, face = "bold"),
    panel.grid.major.y = element_blank()
  )

# Guardar figura en alta resolución
save_plot(p_boruta, file.path(out_dir, "boruta_importance_clean.png"), w = 8, h = 7)

# Abrir automáticamente en macOS
if (Sys.info()["sysname"] == "Darwin") {
  system2("open", file.path(out_dir, "boruta_importance_clean.png"))
}

cat("✅ Gráfico limpio de importancia guardado en outputs/boruta/boruta_importance_clean.png\n")

source("1_meta_analysis/00_setup.R")
library(tibble)
library(ggplot2)

out_dir <- file.path(OUT_DIR, "boruta")
sel_file <- file.path(out_dir, "boruta_selected_otu.csv")

if (!file.exists(sel_file)) {
  stop("❌ No se encontró boruta_selected_otu.csv. Ejecuta primero 05_boruta.R")
}

sel <- readr::read_csv(sel_file, show_col_types = FALSE) %>%
  tibble::column_to_rownames("Sample")

#should be numeric — but if Boruta's output CSV includes the Group or covariate columns (it shouldn't, but check), prcomp will error. Add a guard:
# sel <- sel[, sapply(sel, is.numeric), drop = FALSE]

ps <- readRDS(file.path(OUT_DIR, "models/physeq_filtered.rds"))
meta <- as.data.frame(sample_data(ps))
group_var <- cfg$columns$group

# 1. PCA
pca_res <- prcomp(sel, scale. = TRUE)
pca_df <- as.data.frame(pca_res$x[, 1:2])
pca_df$Sample <- rownames(pca_df)

raw_group_pca <- meta[rownames(pca_df), group_var]
if (is.data.frame(raw_group_pca) || inherits(raw_group_pca, "sample_data")) {
  raw_group_pca <- raw_group_pca[[1]]
}
pca_df$Group <- as.factor(as.character(raw_group_pca))

color_values <- unlist(cfg$groups$colors)

p_pca <- ggplot(pca_df, aes(x = PC1, y = PC2, color = Group)) +
  geom_point(size = 3, alpha = 0.9) +
  theme_minimal(base_size = 14) +
  labs(
    title = "PCA using Boruta-selected ASVs",
    x = paste0("PC1 (", round(summary(pca_res)$importance[2, 1] * 100, 1), "%)"),
    y = paste0("PC2 (", round(summary(pca_res)$importance[2, 2] * 100, 1), "%)")
  ) +
  scale_color_manual(values = color_values)

save_plot(p_pca, file.path(out_dir, "boruta_pca.png"), w = 6, h = 5)

# 2. Heatmap
required <- c("pheatmap", "RColorBrewer")
miss <- required[!vapply(required, requireNamespace, logical(1), quietly = TRUE)]
if (length(miss) > 0) {
  stop("Faltan paquetes para el heatmap: ", paste(miss, collapse = ", "))
}

library(pheatmap)
library(RColorBrewer)

otu_sd <- apply(sel, 2, sd, na.rm = TRUE)
sel2 <- sel[, otu_sd != 0, drop = FALSE]

otu_z <- t(scale(sel2))
otu_z[!is.finite(otu_z)] <- NA
otu_z <- otu_z[complete.cases(otu_z), , drop = FALSE]

sample_ids <- colnames(otu_z)

# Safe extraction for annotations
raw_group_hm <- meta[sample_ids, group_var]
if (is.data.frame(raw_group_hm) || inherits(raw_group_hm, "sample_data")) {
  raw_group_hm <- raw_group_hm[[1]]
}

annotation_col <- data.frame(Group = factor(as.character(raw_group_hm)))
rownames(annotation_col) <- sample_ids

sample_order <- sample_ids[order(annotation_col$Group)]
otu_z2 <- otu_z[, sample_order, drop = FALSE]
ann2 <- annotation_col[sample_order, , drop = FALSE]

ctrl_name <- cfg$groups$levels[1]
ctrl_samples <- rownames(ann2)[ann2$Group == ctrl_name]
if (length(ctrl_samples) > 0) {
  ctrl_means <- rowMeans(otu_z2[, ctrl_samples, drop = FALSE], na.rm = TRUE)
  otu_order <- order(ctrl_means, decreasing = TRUE)
  otu_z3 <- otu_z2[otu_order, , drop = FALSE]
} else {
  otu_z3 <- otu_z2
}

ann_colors <- list(Group = color_values)
breaks_list <- seq(-0.7, 0.7, length.out = 101)

png(file.path(out_dir, "boruta_heatmap_samples.png"), width = 1800, height = 1200, res = 200)
pheatmap(
  otu_z3,
  annotation_col = ann2,
  annotation_colors = ann_colors,
  show_colnames = FALSE,
  show_rownames = TRUE,
  scale = "none",
  color = colorRampPalette(rev(brewer.pal(n = 7, name = "RdBu")))(100),
  breaks = breaks_list,
  cluster_cols = FALSE,
  cluster_rows = FALSE,
  main = "Heatmap of Boruta-selected ASVs (Z-score)"
)
dev.off()

cat("✅ PCA y Heatmap de Boruta generados con éxito en outputs/boruta/.\n\n")

# DESCRIPCIÓN: Análisis db-RDA y selección de modelos basados en 
#              distancias con ajuste de covariables, envfit taxonómico
#              y generación de gráficos descriptivos (PDF + PNG).
# ENVIRONMENT: conda activate qiime2-amplicon-2024.5
# USAGE: Rscript 1_meta_analysis/07_dbrda_envfit_fix.R

source("1_meta_analysis/00_setup.R")
source("utils/utils_phyloseq.R")
source("utils/utils_io.R")
source("utils/utils_plot.R")

library(phyloseq)
library(vegan)
library(dplyr)
library(tibble)
library(ggplot2)

# 1. Primero cargamos la configuración para que 'cfg' esté disponible
cfg <- yaml::read_yaml("config/config.yaml")

out_dir <- file.path(cfg$outputs$dir, "dbrda")
dir_create(out_dir)

# ============================================================
# CHECKPOINT: Omitir computación pesada si ya existen los outputs
# ============================================================
final_target_file <- file.path(out_dir, "dbrda_model.rds")

if (file.exists(final_target_file)) {
  cat(sprintf("⏩ [SALTADO] Los resultados de db-RDA ya existen en: %s\n", final_target_file))
  cat("   (Borra 'outputs/dbrda/dbrda_model.rds' si deseas regenerar modelos y gráficos).\n\n")
} else {
  
  # ============================================================
  # SELECCIÓN DE MODELO: FILTRADO VS UNFILTERED
  # (Descomenta la opción que prefieras usar)
  # ============================================================
  
  # OPCIÓN A: Usar el objeto filtrado por defecto (Alineado con Boruta/LEfSe/MaAsLin2)
  physeq <- readRDS(file.path(cfg$outputs$dir, "models/physeq_filtered.rds"))
  model_type_used <- "FILTERED"
  
  # OPCIÓN B: Usar el objeto crudo/unfiltered (Comenta la línea de arriba y descomenta esta)
  # physeq <- build_phyloseq()
  # model_type_used <- "UNFILTERED"
  
  pseq <- physeq
  cat(sprintf("✅ RUNNING %s MODEL.\n", model_type_used))
  
  dist_method <- cfg$analysis$beta$distance
  bray_dist <- phyloseq::distance(pseq, method = dist_method)
  
  # Convertir explícitamente a data.frame plano para compatibilidad con vegan
  meta <- as.data.frame(as(sample_data(pseq), "data.frame"), stringsAsFactors = FALSE)
  group_col <- cfg$columns$group
  meta[[group_col]] <- factor(meta[[group_col]], levels = cfg$groups$levels)
  
  # Sincronizar muestras con labels en minúscula por seguridad
  common_samples <- intersect(rownames(meta), labels(bray_dist))
  meta <- meta[common_samples, , drop = FALSE]
  bray_dist <- as.dist(as.matrix(bray_dist)[common_samples, common_samples])
  
  # ============================================================
  # SELECCIÓN ESCALONADA (STEPWISE SCOPE)
  # ============================================================
  scope_covs <- cfg$differential$dbrda$scope_covariates
  mod0 <- stats::as.formula(paste("bray_dist ~", group_col))
  mod1 <- stats::as.formula(paste("bray_dist ~", group_col, "+", paste(scope_covs, collapse = " + ")))
  
  cap0 <- capscale(mod0, data = meta)
  cap1 <- capscale(mod1, data = meta)
  
  step_model <- ordiR2step(cap0, scope = formula(cap1),
                           direction = "forward", R2scope = TRUE, 
                           pstep = cfg$differential$dbrda$permutations)
  
  saveRDS(step_model, file.path(out_dir, "ordiR2step_model.rds"))
  writeLines(capture.output(step_model), con = file.path(out_dir, "ordiR2step_log.txt"))
  
  # ============================================================
  # MODELO CONDICIONADO db-RDA Y ESTADÍSTICAS
  # ============================================================
  cond_covs <- cfg$differential$dbrda$condition_covariates
  if (length(cond_covs) > 0) {
    dbRDA_formula <- stats::as.formula(paste0("bray_dist ~ ", group_col, " + Condition(", paste(cond_covs, collapse = ", "), ")"))
    cond_text <- paste(cond_covs, collapse = ", ")
  } else {
    dbRDA_formula <- stats::as.formula(paste0("bray_dist ~ ", group_col))
    cond_text <- "None"
  }
  
  dbRDA_model <- capscale(dbRDA_formula, data = meta)
  saveRDS(dbRDA_model, file.path(out_dir, "dbrda_model.rds"))
  
  anova_all <- anova(dbRDA_model, permutations = cfg$differential$dbrda$permutations)
  anova_margin <- anova(dbRDA_model, by = "margin", permutations = cfg$differential$dbrda$permutations)
  
  write_csv_safe(as.data.frame(anova_all) %>% tibble::rownames_to_column("Term"), file.path(out_dir, "dbrda_anova_all.csv"))
  write_csv_safe(as.data.frame(anova_margin) %>% tibble::rownames_to_column("Term"), file.path(out_dir, "dbrda_anova_margin.csv"))
  
  # R² y R² Ajustado
  r2 <- RsquareAdj(dbRDA_model)
  writeLines(c(
    paste0("R2:           ", r2$r.squared),
    paste0("Adjusted R2:  ", r2$adj.r.squared)
  ), con = file.path(out_dir, "dbrda_r2.txt"))
  
  # ============================================================
  # ENVFIT: FILTRO DE PREVALENCIA Y VECTORES TAXONÓMICOS
  # ============================================================
  otu_mat <- as(otu_table(pseq), "matrix")
  if (!taxa_are_rows(pseq)) otu_mat <- t(otu_mat)
  otu_mat <- t(otu_mat)[common_samples, , drop = FALSE]
  
  prev <- apply(otu_mat, 2, function(x) sum(x > 0) / length(x))
  otu_f <- otu_mat[, prev > cfg$differential$dbrda$prevalence_min, drop = FALSE]
  
  fit <- envfit(dbRDA_model, otu_f,
                perm  = cfg$differential$dbrda$permutations,
                p.max = 1)
  
  fit_scores <- as.data.frame(scores(fit, display = "vectors"))
  fit_scores$pval <- fit$vectors$pvals
  fit_scores$r2_vec <- fit$vectors$r
  fit_scores$OTU_ID <- rownames(fit_scores)
  
  stopifnot(nrow(fit_scores) == length(fit$vectors$pvals))
  
  fit_sig <- fit_scores[!is.na(fit_scores$pval) & fit_scores$pval < 0.05, , drop = FALSE]
  
  tax_df <- as.data.frame(tax_table(pseq)) %>% tibble::rownames_to_column("OTU_ID")
  fit_sig_annot <- merge(fit_sig, tax_df, by = "OTU_ID") %>%
    arrange(pval, desc(r2_vec))
  
  write_csv_safe(fit_sig, file.path(out_dir, "envfit_significant_otus.csv"))
  write_csv_safe(fit_sig_annot, file.path(out_dir, "envfit_significant_otus_with_taxonomy.csv"))
  
  # ============================================================
  # EXTRACCIÓN DE MÉTRICAS PARA VISUALIZACIÓN DESCRIPTIVA
  # ============================================================
  sites_df <- as.data.frame(scores(dbRDA_model, display = "sites", choices = c(1, 2)))
  axis_names <- colnames(sites_df) 
  colnames(sites_df) <- c("Axis1", "Axis2")
  sites_df$Sample <- rownames(sites_df)
  sites_df$Group <- meta[sites_df$Sample, group_col]
  
  eig_summary <- summary(dbRDA_model)$cont$importance
  var_ax1 <- round(eig_summary["Proportion Explained", axis_names[1]] * 100, 2)
  var_ax2 <- round(eig_summary["Proportion Explained", axis_names[2]] * 100, 2)
  
  anova_df <- as.data.frame(anova_margin)
  f_val <- round(anova_df[1, "F"], 3)
  p_val <- anova_df[1, "Pr(>F)"]
  p_text <- ifelse(p_val < 0.001, "p < 0.001", sprintf("p = %.4f", p_val))
  
  r2_pct <- round(r2$r.squared * 100, 2)
  adj_r2_pct <- round(r2$adj.r.squared * 100, 2)
  
  group_counts <- table(sites_df$Group)
  counts_str <- paste(names(group_counts), group_counts, sep = ": n=", collapse = " | ")
  
  stats_box <- sprintf(
    "Modelo: ~ %s + Condition(%s)\nDistancia: %s | Permutaciones: %d\nR² = %.2f%% (Adj. R² = %.2f%%)\nANOVA (%s): F = %.3f, %s\nASVs sig. envfit (p < 0.05): %d / %d",
    group_col, cond_text, toupper(dist_method), cfg$differential$dbrda$permutations,
    r2_pct, adj_r2_pct, group_col, f_val, p_text, nrow(fit_sig_annot), ncol(otu_f)
  )
  
  colors_map <- unlist(cfg$groups$colors)
  
  # ============================================================
  # GRÁFICO 1: ORDENACIÓN DE MUESTRAS (SITES) CON ESTADÍSTICAS
  # ============================================================
  p_sites <- ggplot(sites_df, aes(x = Axis1, y = Axis2, color = Group, fill = Group)) +
    geom_hline(yintercept = 0, linetype = "dashed", color = "grey65", linewidth = 0.4) +
    geom_vline(xintercept = 0, linetype = "dashed", color = "grey65", linewidth = 0.4) +
    stat_ellipse(geom = "polygon", alpha = 0.12, level = 0.95, linewidth = 0.6, show.legend = FALSE) +
    geom_point(size = 2.8, alpha = 0.85) +
    scale_color_manual(values = colors_map) +
    scale_fill_manual(values = colors_map) +
    annotate(
      "label",
      x = -Inf, y = Inf,
      label = stats_box,
      hjust = -0.04, vjust = 1.08,
      size = 3.1,
      fill = alpha("white", 0.88),
      color = "grey15",
      label.size = 0.3
    ) +
    labs(
      title = "Distance-based Redundancy Analysis (db-RDA)",
      subtitle = sprintf("Muestras totales: N = %d (%s) | Elipses al 95%% de confianza", nrow(sites_df), counts_str),
      x = sprintf("%s (%.2f%% varianza explicada / constrained)", axis_names[1], var_ax1),
      y = sprintf("%s (%.2f%% varianza residual / unconstrained)", axis_names[2], var_ax2),
      color = "Grupo",
      caption = "Generado por Script 07 (vegan::capscale) | Covariables ajustadas mediante Condition()"
    ) +
    theme_bw(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", size = 14),
      plot.subtitle = element_text(size = 10, color = "grey30"),
      legend.position = "top",
      panel.grid.minor = element_blank()
    )
  
  ggsave(file.path(out_dir, "dbrda_sites.pdf"), plot = p_sites, width = 8.5, height = 6.5)
  ggsave(file.path(out_dir, "dbrda_sites.png"), plot = p_sites, width = 8.5, height = 6.5, dpi = 300)
  
  # ============================================================
  # GRÁFICO 2 (BONUS): BIPLOT CON TOP 10 VECTORES ENVFIT
  # ============================================================
  if (nrow(fit_sig_annot) > 0) {
    top_vec <- head(fit_sig_annot, 10)
    colnames(top_vec)[colnames(top_vec) == axis_names[1]] <- "Vec1"
    colnames(top_vec)[colnames(top_vec) == axis_names[2]] <- "Vec2"
    
    top_vec$TaxLabel <- ifelse(
      !is.na(top_vec$Genus) & top_vec$Genus != "" & top_vec$Genus != "Unassigned",
      paste0(gsub("^g__", "", top_vec$Genus), " (", substr(top_vec$OTU_ID, 1, 5), ")"),
      paste0("ASV_", substr(top_vec$OTU_ID, 1, 6))
    )
    
    mult <- min(
      (max(sites_df$Axis1) - min(sites_df$Axis1)) / (max(abs(top_vec$Vec1)) * 2.2),
      (max(sites_df$Axis2) - min(sites_df$Axis2)) / (max(abs(top_vec$Vec2)) * 2.2)
    )
    if (!is.finite(mult) || mult <= 0) mult <- 1
    
    top_vec$xend <- top_vec$Vec1 * mult
    top_vec$yend <- top_vec$Vec2 * mult
    
    p_biplot <- p_sites +
      geom_segment(
        data = top_vec,
        aes(x = 0, y = 0, xend = xend, yend = yend),
        inherit.aes = FALSE,
        arrow = arrow(length = unit(0.22, "cm")),
        color = "#1b7837",
        linewidth = 0.65,
        alpha = 0.85
      ) +
      geom_text(
        data = top_vec,
        aes(x = xend * 1.08, y = yend * 1.08, label = TaxLabel),
        inherit.aes = FALSE,
        size = 3.0,
        fontface = "italic",
        color = "#00441b"
      ) +
      labs(
        title = "db-RDA Biplot con Vectores Taxonómicos Significativos (envfit)",
        subtitle = sprintf("Top %d ASVs asociados (p < 0.05) | R² = %.2f%% (%s)", nrow(top_vec), r2_pct, p_text)
      )
    
    ggsave(file.path(out_dir, "dbrda_biplot_envfit.pdf"), plot = p_biplot, width = 9, height = 7)
    ggsave(file.path(out_dir, "dbrda_biplot_envfit.png"), plot = p_biplot, width = 9, height = 7, dpi = 300)
  }
  
  cat("✅ Script 07 finalizado con éxito. Gráficos guardados en PDF y PNG en outputs/dbrda/.\n")
}

  common_samples <- intersect(colnames(ko_mat_full), colnames(tax_mat2))
  if (length(common_samples) == 0) stop("No common samples between KO and taxa matrices.")
  common_samples <- sort(common_samples)
  
  ko_mat <- ko_mat_full[, common_samples, drop = FALSE]
  tax_mat_aligned <- tax_mat2[, common_samples, drop = FALSE]
  
  # Transformaciones logarítmicas seguras usando dobles corchetes para "function"
  ko_log <- log_transform(ko_mat,
                          base = if (!is.null(cfg[["function"]]$transform$log_base)) cfg[["function"]]$transform$log_base else 2,
                          pseudocount = if (!is.null(cfg[["function"]]$transform$pseudocount)) cfg[["function"]]$transform$pseudocount else 1)

                           base = if (!is.null(cfg[["function"]]$transform$log_base)) cfg[["function"]]$transform$log_base else 2,
                           pseudocount = if (!is.null(cfg[["function"]]$transform$pseudocount)) cfg[["function"]]$transform$pseudocount else 1)
  
  # ============================================================
  # SCRIPT 10 OPTIMIZADO: MATRICIAL Y VECTORIAL (MÉTODO RÁPIDO)
  # ============================================================
  
  # 1. Matriz de correlación de Spearman vectorizada
  cor_mat <- cor(t(tax_log), t(ko_log), 
                 method = cfg[["function"]]$correlation$method, 
                 use = "pairwise.complete.obs")
  
  # 2. Tamaño de muestra (n) para grados de libertad
  n <- ncol(tax_log)
  df <- n - 2
  
  # 3. Cálculo vectorizado de estadísticos t
  r2 <- cor_mat^2
  r2[r2 >= 1] <- 0.9999999 
  t_stat <- cor_mat * sqrt(df / (1 - r2))
  
  # 4. Cálculo vectorizado de p-valores usando distribución t de Student
  p_mat <- matrix(2 * pt(-abs(t_stat), df = df), 
                  nrow = nrow(cor_mat), 
                  ncol = ncol(cor_mat), 
                  dimnames = dimnames(cor_mat))
  
  # 5. Ajuste FDR (Benjamini-Hochberg) y conversión a tablas largas
  q_vec <- p.adjust(as.vector(p_mat), method = cfg[["function"]]$correlation$fdr_method)
  q_mat <- matrix(q_vec, nrow = nrow(p_mat), ncol = ncol(p_mat), dimnames = dimnames(p_mat))
  
  cor_long <- as.data.frame(as.table(cor_mat)) %>% rename(Taxon = Var1, KO = Var2, rho = Freq)
  p_long   <- as.data.frame(as.table(p_mat))   %>% rename(Taxon = Var1, KO = Var2, p = Freq)
  q_long   <- as.data.frame(as.table(q_mat))   %>% rename(Taxon = Var1, KO = Var2, q = Freq)
  
  # Combinar resultados en la tabla global de correlaciones
  cor_full <- cor_long %>%
    left_join(p_long, by = c("Taxon", "KO")) %>%
    left_join(q_long, by = c("Taxon", "KO"))
  
  write_csv_safe(cor_full, file.path(out_dir, "tables/taxon_ko_cor_all.csv"))
  
  # Filtrar aristas significativas según umbrales del config.yaml
  edge_df <- cor_full %>%
    filter(!is.na(rho) & !is.na(p) & abs(rho) >= cfg[["function"]]$correlation$abs_rho_min & p <= cfg[["function"]]$correlation$p_max)
  
  write_csv_safe(edge_df, file.path(out_dir, "tables/taxon_ko_edges_filtered.csv"))
  
  cat("¡Script 10 vectorizado finalizado con éxito en segundos!\n")
}

  dplyr::select(Sample, Group) %>% 
  dplyr::mutate(Set_for_taxa = "Set1")

write_csv_safe(meta_taxa, file.path(out_tables, "metadata_for_taxa.csv"))

tax_df <- as.data.frame(as(tax_table(ps), "matrix")) %>% tibble::rownames_to_column("ASV")
tax_filled <- tax_df %>%
  dplyr::mutate(
    Set_taxon = paste("Set1", Genus, sep = "_"),
    Taxa_name = ifelse(is.na(Genus) | Genus == "", "g__Unassigned", Genus)
  )

write_csv_safe(tax_filled, file.path(out_tables, "picrust_taxonomy_filled.csv"))

# 4. KO Fold Change
ko_full <- read_tsv("outputs/tables/merged_KO_metagenome.tsv", show_col_types = FALSE)
colnames(ko_full)[1] <- "KO"

ko_long <- ko_full %>%
  pivot_longer(cols = -KO, names_to = "Sample", values_to = "Abundance") %>%
  left_join(meta_taxa, by = "Sample") %>%
  group_by(KO, Group) %>%
  summarise(mean_ab = mean(Abundance, na.rm = TRUE), .groups = "drop") %>%
  pivot_wider(names_from = Group, values_from = mean_ab) %>%
  mutate(
    fc = Insomnia - Control,
    Strength = case_when(fc > 0 ~ "Upregulated", fc < 0 ~ "Downregulated", TRUE ~ "Unchanged"),
    ko_function = KO
  )

write_csv_safe(ko_long, file.path(out_tables, "KO_FC_group.csv"))

  tibble::rownames_to_column("ASV") %>%

  left_join(tax_filled %>% select(ASV, Genus), by = "ASV") %>%
  filter(!is.na(Genus) & Genus != "")

# Aggregate counts per sample and genus

  group_by(sample, Genus) %>%
  summarise(norm_taxon_function_contrib = sum(count), .groups = "drop") %>%
  rename(taxon = Genus)

contrib_df <- read_csv(contrib_path, show_col_types = FALSE)
meta_taxa <- read_csv(meta_path, show_col_types = FALSE)

# 2. Perform differential contribution calculations
cat("📊 Processing contribution shifts between groups...\n")

# Ensure metadata has lowercase 'sample' for proper joining
if ("Sample" %in% colnames(meta_taxa) && !"sample" %in% colnames(meta_taxa)) {
  meta_taxa <- meta_taxa %>% rename(sample = Sample)
}

# Merge contribution data with metadata groups
contrib_merged <- contrib_df %>%
  left_join(meta_taxa, by = "sample")

# Calculate mean contribution per group for each KO and taxon
contrib_summary <- contrib_merged %>%
  group_by(KO, taxon, Group) %>%
  summarise(mean_contrib = mean(norm_taxon_function_contrib, na.rm = TRUE), .groups = "drop") %>%
  pivot_wider(names_from = Group, values_from = mean_contrib, values_fill = 0)

# Compute contribution change (Insomnia vs Control)
if (all(c("Insomnia", "Control") %in% colnames(contrib_summary))) {
  contrib_summary <- contrib_summary %>%
    mutate(contrib_diff = Insomnia - Control)
} else {
  cols <- setdiff(colnames(contrib_summary), c("KO", "taxon"))
  if (length(cols) >= 2) {
    contrib_summary <- contrib_summary %>%
      mutate(contrib_diff = .data[[cols[2]]] - .data[[cols[1]]])
  } else {
    contrib_summary <- contrib_summary %>% mutate(contrib_diff = 0)
  }
}

# 3. Save final results
output_file <- file.path(out_contrib_dir, "differential_functional_contributions.csv")
write_csv_safe(contrib_summary, output_file)

cat(sprintf("✅ Success! Differential functional contribution table saved to: %s\n", output_file))

suppressPackageStartupMessages({
  library(readr)
  library(dplyr)
})

# Lista de las tablas principales (rutas relativas a la raíz del proyecto 16s_insomnia/)
archivos_clave <- c(
  "Diversidad Alfa"                  = "outputs/tables/alpha_diversity.csv",
  "Ratio Firmicutes/Bacteroidota"    = "outputs/tables/fb_ratio.csv",
  "ASVs Seleccionados (Boruta)"      = "outputs/boruta/boruta_selected_otus_with_taxonomy.csv",
  "Biomarcadores LEfSe"             = "outputs/lefse/lefse_sig_with_otu.csv",
  "Correlaciones Taxón-KO (Edges)"   = "outputs/function/tables/taxon_ko_edges_filtered.csv",
  "Fold-Change de KOs"               = "outputs/tables/KO_FC_group.csv"
)

cat("=====================================================================\n")
cat("INSPECCIÓN AUTOMATIZADA DE RESULTADOS CLAVE\n")
cat("=====================================================================\n\n")

for (nombre in names(archivos_clave)) {
  ruta <- archivos_clave[nombre]
  
  cat(sprintf("---------------------------------------------------------------------\n"))
  cat(sprintf("📁 TABLA: %s\n", nombre))
  cat(sprintf("📂 Archivo: %s\n", ruta))
  cat(sprintf("---------------------------------------------------------------------\n"))
  
  if (file.exists(ruta)) {
    # Cargar la tabla de forma segura
    df <- read_csv(ruta, show_col_types = FALSE)
    
    # Imprimir dimensiones y estructura básica
    cat(sprintf("Dimensiones: %d filas × %d columnas\n\n", nrow(df), ncol(df)))
    
    cat("Primeras 3 filas:\n")
    print(head(df, 3))
    cat("\n")
    
    # Mostrar un summary rápido
    if (nrow(df) > 0) {
      cat("Resumen estadístico básico:\n")
      print(summary(df))
    }
    
  } else {
    cat("⚠️ [AVISO] El archivo no se encontró en esta ruta (verifica si el módulo previo ya corrió).\n")
  }
  
  cat("\n\n")
}

cat("=====================================================================\n")
cat("¡Inspección finalizada con éxito!\n")
cat("=====================================================================\n")
